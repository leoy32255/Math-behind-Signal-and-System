% Remove empty rows at the start or end (e.g. from a trailing \\).
\bool_new:N \l_lralign_trim_stop_bool
\cs_new_protected:Npn \lralign_trim_rows:N #1
{
    \bool_set_false:N \l_lralign_trim_stop_bool
    \bool_until_do:nn { \l_lralign_trim_stop_bool }
    {
        \seq_if_empty:NTF #1
        { \bool_set_true:N \l_lralign_trim_stop_bool }
        {
            \seq_get_left:NN #1 \l_tmpa_tl
            \tl_if_blank:VTF \l_tmpa_tl
            { \seq_pop_left:NN #1 \l_tmpa_tl }
            { \bool_set_true:N \l_lralign_trim_stop_bool }
        }
    }
    \bool_set_false:N \l_lralign_trim_stop_bool
    \bool_until_do:nn { \l_lralign_trim_stop_bool }
    {
        \seq_if_empty:NTF #1
        { \bool_set_true:N \l_lralign_trim_stop_bool }
        {
            \seq_get_right:NN #1 \l_tmpa_tl
            \tl_if_blank:VTF \l_tmpa_tl
            { \seq_pop_right:NN #1 \l_tmpa_tl }
            { \bool_set_true:N \l_lralign_trim_stop_bool }
        }
    }
}

\cs_new_protected:Npn \lralign_measure_row:nNNNN #1#2#3#4#5
{
    \seq_set_split:Nnn \l_lralign_cell_seq { & } { #1 }
    \int_case:nnF { \seq_count:N \l_lralign_cell_seq }
    {
        { 1 }
        {
            \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
            \hbox_set:Nn \l_lralign_measure_box
            { $\displaystyle \l_tmpa_tl$ }
            \dim_set:Nn #4
            { \dim_max:nn { #4 } { \box_wd:N \l_lralign_measure_box } }
        }
        { 2 }
        {
            \bool_set_true:N #5
            \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
            \seq_pop_left:NN \l_lralign_cell_seq \l_tmpb_tl
            \hbox_set:Nn \l_lralign_measure_box
            { $\displaystyle \l_tmpa_tl$ }
            \dim_set:Nn #2
            { \dim_max:nn { #2 } { \box_wd:N \l_lralign_measure_box } }
            \hbox_set:Nn \l_lralign_measure_box
            { $\displaystyle {} \l_tmpb_tl$ }
            \dim_set:Nn #3
            { \dim_max:nn { #3 } { \box_wd:N \l_lralign_measure_box } }
        }
    }
    { \msg_error:nn { lralign } { too-many-alignment-tabs } }
}

% Append side cell in side-by-side mode.
% #1 = color, #2 = row text, #3 = pre col dim, #4 = post col dim, #5 = total side dim, #6 = leading hspace dim
\cs_new_protected:Npn \lralign_append_side_sbs:nnNNNN #1#2#3#4#5#6
{
    \dim_compare:nNnF { #6 } = { 0pt }
    {
        \tl_put_right:Nx \l_lralign_final_body_tl
        { \exp_not:N \hspace { \dim_use:N #6 } }
    }
    \tl_if_blank:nTF { #2 }
    {
        \tl_put_right:Nx \l_lralign_final_body_tl
        {
            \exp_not:N \mathmakebox [ \dim_use:N #3 ] [r] {}
            &
            \exp_not:N \mathmakebox [ \dim_use:N #4 ] [l] {}
        }
    }
    {
        \seq_set_split:Nnn \l_lralign_cell_seq { & } { #2 }
        \int_case:nnF { \seq_count:N \l_lralign_cell_seq }
        {
            { 1 }
            {
                \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
                \dim_compare:nNnTF { #4 } = { 0pt }
                {
                    \tl_put_right:Nx \l_lralign_final_body_tl
                    {
                        \exp_not:N \mathmakebox [ \dim_use:N #5 ] [c]
                        { \exp_not:N \color { #1 } \exp_not:V \l_tmpa_tl }
                        &
                    }
                }
                {
                    \tl_put_right:Nx \l_lralign_final_body_tl
                    {
                        \exp_not:N \mathmakebox [ \dim_use:N #3 ] [l]
                        {
                            \exp_not:N \mathmakebox [0pt] [l]
                            {
                                \exp_not:N \mathmakebox [ \dim_use:N #5 ] [c]
                                { \exp_not:N \color { #1 } \exp_not:V \l_tmpa_tl }
                            }
                        }
                        &
                        \exp_not:N \mathmakebox [ \dim_use:N #4 ] [l] {}
                    }
                }
            }
            { 2 }
            {
                \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
                \seq_pop_left:NN \l_lralign_cell_seq \l_tmpb_tl
                \tl_put_right:Nx \l_lralign_final_body_tl
                {
                    \exp_not:N \mathmakebox [ \dim_use:N #3 ] [r]
                    { \exp_not:N \color { #1 } \exp_not:V \l_tmpa_tl }
                    &
                    \exp_not:N \mathmakebox [ \dim_use:N #4 ] [l]
                    { \exp_not:N \color { #1 } {} \exp_not:V \l_tmpb_tl }
                }
            }
        }
        { \msg_error:nn { lralign } { too-many-alignment-tabs } }
    }
}

\cs_new_protected:Npn \lralign_process:nn #1#2
{
    \tl_set:Nn \l_lralign_label_tl { #1 }
    % Split input at \switch
    \seq_set_split:Nnn \l_lralign_parts_seq { \switch } { #2 }
    \seq_pop_left:NN \l_lralign_parts_seq \l_lralign_left_body_tl
    \seq_if_empty:NTF \l_lralign_parts_seq
    { \tl_clear:N \l_lralign_right_body_tl }
    { \seq_pop_left:NN \l_lralign_parts_seq \l_lralign_right_body_tl }

    % Split into rows and trim leading/trailing blank rows
    \seq_set_split:NnV \l_lralign_left_rows_seq { \\ } \l_lralign_left_body_tl
    \seq_set_split:NnV \l_lralign_right_rows_seq { \\ } \l_lralign_right_body_tl
    \lralign_trim_rows:N \l_lralign_left_rows_seq
    \lralign_trim_rows:N \l_lralign_right_rows_seq

    \int_set:Nn \l_lralign_left_count_int { \seq_count:N \l_lralign_left_rows_seq }
    \int_set:Nn \l_lralign_right_count_int { \seq_count:N \l_lralign_right_rows_seq }
    \int_set:Nn \l_lralign_max_rows_int
    { \int_max:nn { \l_lralign_left_count_int } { \l_lralign_right_count_int } }

    % Measure natural width of each side
    \lralign_measure_rows:NNNNN
        \l_lralign_left_rows_seq
        \l_lralign_left_pre_max_dim
        \l_lralign_left_post_max_dim
        \l_lralign_left_cent_max_dim
        \l_lralign_left_has_amp_bool
    \lralign_measure_rows:NNNNN
        \l_lralign_right_rows_seq
        \l_lralign_right_pre_max_dim
        \l_lralign_right_post_max_dim
        \l_lralign_right_cent_max_dim
        \l_lralign_right_has_amp_bool

    \dim_set:Nn \l_lralign_left_width_dim
    {
        \dim_max:nn
            { \l_lralign_left_pre_max_dim + \l_lralign_left_post_max_dim }
            { \l_lralign_left_cent_max_dim }
    }
    \dim_set:Nn \l_lralign_right_width_dim
    {
        \dim_max:nn
            { \l_lralign_right_pre_max_dim + \l_lralign_right_post_max_dim }
            { \l_lralign_right_cent_max_dim }
    }

    % Measure safe equation tag margin: max( (\thesection.99), (\theequation) ) + 1.5em
    \bool_if:NTF \l_lralign_numbered_bool
    {
        \hbox_set:Nn \l_lralign_measure_box { \normalfont\normalsize (\thesection.99) }
        \dim_set:Nn \l_lralign_tag_dim { \box_wd:N \l_lralign_measure_box }
        \hbox_set:Nn \l_lralign_measure_box { \normalfont\normalsize (\theequation) }
        \dim_set:Nn \l_lralign_tag_dim
        { \dim_max:nn { \l_lralign_tag_dim } { \box_wd:N \l_lralign_measure_box } + 1.5em }
    }
    {
        \dim_zero:N \l_lralign_tag_dim
    }

    \dim_set:Nn \l_lralign_gap_dim { 2.0em }
    \dim_set:Nn \l_lralign_slack_dim
    {
        \linewidth - 2 \l_lralign_tag_dim
        - \l_lralign_left_width_dim - \l_lralign_right_width_dim
    }

    \tl_clear:N \l_lralign_final_body_tl

    \dim_compare:nNnTF { \l_lralign_slack_dim } < { \l_lralign_gap_dim }
    {
        % Stacked mode: frequency (blue) on top, angular frequency (red) below
        \bool_if:nTF { \l_lralign_left_has_amp_bool || \l_lralign_right_has_amp_bool }
        {
            \int_step_inline:nn { \l_lralign_left_count_int }
            {
                \seq_pop_left:NN \l_lralign_left_rows_seq \l_tmpa_tl
                \lralign_append_stacked_align_row:nV { freqcolor } \l_tmpa_tl
                \int_compare:nNnTF { ##1 } = { \l_lralign_left_count_int }
                {
                    \int_compare:nNnT { \l_lralign_right_count_int } > { 0 }
                    { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\[4pt] } }
                }
                { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\ } }
            }
            \int_step_inline:nn { \l_lralign_right_count_int }
            {
                \seq_pop_left:NN \l_lralign_right_rows_seq \l_tmpa_tl
                \lralign_append_stacked_align_row:nV { angcolor } \l_tmpa_tl
                \int_compare:nNnF { ##1 } = { \l_lralign_right_count_int }
                { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\ } }
            }
            \bool_if:NTF \l_lralign_numbered_bool
            {
                \lralign_emit_label:
                \lralign_exec_align:V \l_lralign_final_body_tl
            }
            {
                \lralign_exec_align_star:V \l_lralign_final_body_tl
            }
        }
        {
            \int_step_inline:nn { \l_lralign_left_count_int }
            {
                \seq_pop_left:NN \l_lralign_left_rows_seq \l_tmpa_tl
                \lralign_append_stacked_gather_row:nV { freqcolor } \l_tmpa_tl
                \int_compare:nNnTF { ##1 } = { \l_lralign_left_count_int }
                {
                    \int_compare:nNnT { \l_lralign_right_count_int } > { 0 }
                    { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\[4pt] } }
                }
                { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\ } }
            }
            \int_step_inline:nn { \l_lralign_right_count_int }
            {
                \seq_pop_left:NN \l_lralign_right_rows_seq \l_tmpa_tl
                \lralign_append_stacked_gather_row:nV { angcolor } \l_tmpa_tl
                \int_compare:nNnF { ##1 } = { \l_lralign_right_count_int }
                { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\ } }
            }
            \bool_if:NTF \l_lralign_numbered_bool
            {
                \lralign_emit_label:
                \lralign_exec_gather:V \l_lralign_final_body_tl
            }
            {
                \lralign_exec_gather_star:V \l_lralign_final_body_tl
            }
        }
    }
    {
        % Side-by-side mode: compute dynamic comfortable gap between 2.0em and 4.2em
        \dim_set:Nn \l_lralign_gap_dim
        {
            \dim_min:nn { 4.2em }
            { \dim_max:nn { 2.0em } { 0.35 \l_lralign_slack_dim } }
        }

        % Pre/post widths for each side (supporting pure &, pure centered, or mixed)
        \bool_if:NTF \l_lralign_left_has_amp_bool
        {
            \dim_set:Nn \l_lralign_left_pre_column_dim
            {
                \dim_max:nn { \l_lralign_left_pre_max_dim }
                { ( \l_lralign_left_width_dim + \l_lralign_left_pre_max_dim - \l_lralign_left_post_max_dim ) / 2 }
            }
            \dim_set:Nn \l_lralign_left_post_column_dim
            { \l_lralign_left_width_dim - \l_lralign_left_pre_column_dim }
        }
        {
            \dim_set:Nn \l_lralign_left_pre_column_dim { \l_lralign_left_width_dim }
            \dim_zero:N \l_lralign_left_post_column_dim
        }

        \bool_if:NTF \l_lralign_right_has_amp_bool
        {
            \dim_set:Nn \l_lralign_right_pre_column_dim
            {
                \dim_max:nn { \l_lralign_right_pre_max_dim }
                { ( \l_lralign_right_width_dim + \l_lralign_right_pre_max_dim - \l_lralign_right_post_max_dim ) / 2 }
            }
            \dim_set:Nn \l_lralign_right_post_column_dim
            { \l_lralign_right_width_dim - \l_lralign_right_pre_column_dim }
        }
        {
            \dim_set:Nn \l_lralign_right_pre_column_dim { \l_lralign_right_width_dim }
            \dim_zero:N \l_lralign_right_post_column_dim
        }

        \int_step_inline:nn { \l_lralign_max_rows_int }
        {
            \seq_pop_left:NN \l_lralign_left_rows_seq \l_tmpa_tl
            \quark_if_no_value:NT \l_tmpa_tl { \tl_clear:N \l_tmpa_tl }
            \lralign_append_side_sbs:nVNNNN
                { freqcolor } \l_tmpa_tl
                \l_lralign_left_pre_column_dim
                \l_lralign_left_post_column_dim
                \l_lralign_left_width_dim
                \c_zero_dim

            \tl_put_right:Nn \l_lralign_final_body_tl { & }

            \seq_pop_left:NN \l_lralign_right_rows_seq \l_tmpa_tl
            \quark_if_no_value:NT \l_tmpa_tl { \tl_clear:N \l_tmpa_tl }
            \lralign_append_side_sbs:nVNNNN
                { angcolor } \l_tmpa_tl
                \l_lralign_right_pre_column_dim
                \l_lralign_right_post_column_dim
                \l_lralign_right_width_dim
                \l_lralign_gap_dim

            \int_compare:nNnF { ##1 } = { \l_lralign_max_rows_int }
            { \tl_put_right:Nn \l_lralign_final_body_tl { \nonumber \\ } }
        }

        \bool_if:NTF \l_lralign_numbered_bool
        {
            \lralign_emit_label:
            \lralign_exec_alignat:V \l_lralign_final_body_tl
        }
        {
            \lralign_exec_alignat_star:V \l_lralign_final_body_tl
        }
    }
}
